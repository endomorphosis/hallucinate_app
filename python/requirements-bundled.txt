# =============================================================================
# Consolidated runtime requirements for the packaged hallucinate_app
# =============================================================================
# Bundled into the .exe / .dmg / .deb / .rpm and installed into a per-user
# virtual environment on first launch by
# hallucinate_app/node/python_environment_manager.js so the Python MCP daemons
# "just work" out of the box with no manual setup.
#
# The three IPFS packages themselves are installed from BUNDLED LOCAL SOURCE
# (see _installBundledPackages) because the published PyPI wheels are
# incomplete / broken. This file pins the *supporting* dependency layer that
# those packages import at runtime but under-declare in their own metadata.
#
# Daemon entrypoints (hallucinate_app/node/mcp_daemon_manager.js):
#   - ipfs-kit:        python -m ipfs_kit_py.cli mcp start
#   - ipfs-datasets:   python -m uvicorn ipfs_datasets_py.mcp_server.fastapi_service:app
#   - ipfs-accelerate: python -m ipfs_accelerate_py.cli mcp start
# -----------------------------------------------------------------------------

# --- Web / RPC runtime used by the daemons -----------------------------------

uvicorn[standard]>=0.23.0
hypercorn>=0.16.0
# Required by FastAPI to parse multipart/form-data — the ipfs_kit_py and
# ipfs_datasets_py dashboards expose file-upload (UploadFile/Form) endpoints, so
# without this the FastAPI app fails to start with a RuntimeError at route build.
python-multipart>=0.0.6
pydantic>=2.0.0
PyJWT>=2.8.0

# --- Data plane (virtual filesystem + metadata indexes) ----------------------
# numpy pinned <=2.1.3 to satisfy ipfs_datasets_py's declared constraint.

numpy>=1.26.4,<=2.1.3

#     launches if these fail to install on a constrained machine) -------------
